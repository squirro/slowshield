  Clients must trust Caddy's internal root CA. The images have no shell or tar, so read it with an
  ephemeral debug container (Amazon Linux 2023 minimal) that shares the caddy container's process namespace:
    POD=$(kubectl -n {{ .Release.Namespace }} get pod -l app.kubernetes.io/instance={{ .Release.Name }} -o name | head -1)
    kubectl -n {{ .Release.Namespace }} debug -q -i $POD --target=caddy --profile=restricted \
      --image=public.ecr.aws/amazonlinux/amazonlinux:2023-minimal -- \
      cat /proc/1/root/data/caddy/pki/authorities/local/root.crt > slowshield-root.crt
{{- end }}
